using System.ComponentModel.DataAnnotations;

class Cube
{
    private double l, w, h;

    public void setLength(double length)
    { l = length; }

    public void setWidth(double width)
    { w = width; }

    public void setHeight(double height)
    { h = height; }

    public double getVolume()
        { return l * w * h; }
}

class Person
{
    private string fn, ln, g;
    private int a;
    private double w, h;

    public Person
        (
            string firstname,   // parameterized constructor
            string lastname,    // note the parentheses to delimit parameter list
            string gender,
            int age,
            double weight,
            double height
        )
    {
        Console.WriteLine("Person object is being created in memory");  // normally not included
        fn = firstname; // assign input parms to local private vars
        ln = lastname;
        g = gender;
        a = age;
        w = weight;
        h = height;
    }

    public string getFirstName() { return fn; }

    public string getLastName() { return ln; }

    public string getGender() { return g; }

    public int getAge() { return a; }

    public double getHeight() { return h; }

    public double getWidth() { return w; }

    private bool isAFatty()
    {
        if (w / h > 2.25)     // encapsulate this logic
            return true;
        else
            return false;
    }

    public bool failsBMI()
    { return isAFatty(); }

}

        public void AddGrid()
        {
            int counter = 0;

            for (int i = 0; i < 3; i++)
            {

                {
                    TableLayoutPanel child_grid = new TableLayoutPanel();
                    child_grid.Size = new Size(380, 165);
                    child_grid.RowCount = 3;
                    child_grid.ColumnCount = 1;
                    child_grid.CellBorderStyle = TableLayoutPanelCellBorderStyle.None;
                    parent_grid.Controls.Add(child_grid);

                    PictureBox pic = new PictureBox();
                    pic.Name = "pic_" + i + "_" + j;
                    pic.Size = new Size(395, 110);
                    pic.SizeMode = PictureBoxSizeMode.StretchImage;

                    t1.SetToolTip(pic, "Click any image to see details.");

                    // add event handler and delegate via lambda operator
                    if (i == 0 && j == 0) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 0].ID);
                    else if (i == 0 && j == 1) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 1].ID);

                    else if (i == 1 && j == 0) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 3].ID);
                    else if (i == 1 && j == 1) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 4].ID);
                    else if (i == 1 && j == 2) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 5].ID);
                    else if (i == 2 && j == 0) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 6].ID);
                    else if (i == 2 && j == 1) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 7].ID);
                    else if (i == 2 && j == 2) pic.Click += (sender2, e2) => ImageWasClicked(sender2, e2, my_cars[start_index + 8].ID);

                    child_grid.Controls.Add(pic, 0, 0);

                    // add label with text set to manufacturer + model to child grid
                    Label l1 = new Label();
                    l1.Name = "l1_" + i + "_" + j;
                    Font my_font = new Font(FontFamily.GenericSansSerif, 10, FontStyle.Regular);
                    l1.Font = my_font;
                    child_grid.Controls.Add(l1, 0, 1);

                    // add label with text set to price to child grid
                    Label l2 = new Label();
                    l2.Name = "l2_" + i + "_" + j;
                    l2.Font = my_font;
                    child_grid.Controls.Add(l2, 0, 2);

                    counter++;
                }
            }
            AddCarsToGrid(start_index);
        }

        private void AddCarsToGrid(int start_index)
        {
            int counter = 0;

            for (int i = 0; i < 3; i++)
            {
                for (int j = 0; j < 3; j++)
                {
                    foreach (Control c in this.Controls)
                    {
                        if (c is TableLayoutPanel)  // parent
                        {
                            foreach (Control c2 in c.Controls)
                            {
                                if (c2 is TableLayoutPanel)  // child
                                {
                                    foreach(Control c3 in c2.Controls)
                                    {
                                        if (c3 is PictureBox)
                                        {
                                            PictureBox p = (PictureBox)c3;
                                            if (p.Name == "pic_" + i + "_" + j)
                                            {
                                                Image img = Image.FromFile(my_cars[start_index + counter].pics[0].filename);
                                                p.Image = img;
                                            }
                                        }
                                        if (c3 is Label)
                                        {
                                            Label l1 = (Label)c3;
                                            if (l1.Name == "l1_" + i + "_" + j)
                                            {
                                                l1.Text = my_cars[start_index + counter].Manufacturer;
                                            }
                                            else if (l1.Name == "l2_" + i + "_" + j)
                                            {
                                                l1.Text = my_cars[start_index + counter].price.ToString("C", new CultureInfo("en-US"));
                                                counter++;
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        public void ImageWasClicked (object sender, EventArgs e, int index)
        {
            this.Hide();
            CarDetails newPage = new CarDetails(my_cars[index]);
            newPage.ShowDialog();
            this.Close();
        }

        public void ClearSearch(object sender, EventArgs e)
        {
            ClearCars();
            int start_index = random.Next(3);
            AddCarsToGrid(start_index);
        }

        private void DoSearch(object sender, EventArgs e)
        {
            ClearCars();
            ExecuteSearch();
        }

        private void ExecuteSearch()
        {
            WhichCars();
            DisplayMatches();
        }

        private void WhichCars()
        {
            Array.Clear(displayed_cars, 0, displayed_cars.Length);

            for (int i = 0; i < displayed_cars.Length; i++) displayed_cars[i] = -1; // indicates no car

            // for some reason Visual Studio does not recognize the SearchTextbox name so I will force it
            foreach (Control c in this.Controls)
            {
                if (c is TextBox)
                {
                    TextBox txt = (TextBox)c;
                    int counter = 0;
                    foreach (car a_car in my_cars)
                    {
                        if (a_car.Manufacturer.ToUpper().Contains(txt.Text.ToUpper()) ||
                            a_car.Model.ToUpper().Contains(txt.Text.ToUpper()) ||
                            a_car.description.ToUpper().Contains(txt.Text.ToUpper()))
                        {
                            if (counter < 8) // we only have room for 9 cars
                            {
                                displayed_cars[counter] = a_car.ID;
                                counter++;
                            }
                        }
                    }
                }
            }
        }
        
        private void DisplayMatches()
        {
            car this_car = new car();
            if (displayed_cars[0] == - -1) // no matches
                MessageBox.Show("There were no matches to the entered criteria.");
            else
            {
                for (int i = 0; i < displayed_cars.Length; i++)
                {
                    this_car = null;
                    if (displayed_cars[i] >= 0)  // is a valid car ID
                    {
                        foreach (car a_car in my_cars)
                            if (a_car.ID == displayed_cars[i]) this_car = a_car;    // this_car is now pointing to a car to display
                    }
                }

                if (this_car != null)
                {
                    bool applied;
                    foreach(Control a in this.Controls)
                    {
                        applied = false;
                        if (a is TableLayoutPanel) // found parent
                        {
                            foreach(Control b in a.Controls)
                            {
                                if (b is TableLayoutPanel) // found child
                                {
                                    PictureBox p1 = (PictureBox)b.Controls[0]; // we know this because we know order we created objects in child grid
                                    if (p1.Image == null && applied == false)
                                    {
                                        p1.Image = Image.FromFile (this_car.pics[0].filename);
                                        Label l1 = (Label)b.Controls[1];
                                        l1.Text = this_car.Manufacturer + " " + this_car.Model;
                                        Label l2 = (Label)b.Controls[2];
                                        l2.Text = this_car.price.ToString("C", new CultureInfo("en-US"));

                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        private void ClearCars()
        {
            foreach (Control c in this.Controls)
            {
                if (c is TableLayoutPanel)  // parent TableLayoutPanel
                {
                    foreach (Control d in c.Controls)
                    {
                        if (d is TableLayoutPanel)  // child TableLayoutPanel
                        {
                            foreach (Control e in d.Controls)
                            {
                                if (e is PictureBox)
                                {
                                    PictureBox pb1 = (PictureBox)e;
                                    pb1.Image = null;
                                }
                                else if (e is Label)
                                {
                                    Label l1 = (Label)e;
                                    l1.Text = "";
                                }
                            }
                        }
                    }
                }
            }
        }

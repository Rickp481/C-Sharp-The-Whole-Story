    for (int j = 0; j < 5; j++)

        Console.WriteLine("i = {0} and j={1} so i*j={2}", i, j, twodim[i, j]);
    }
}

{
    for (int j = 0; j < 5; j++)
    {
        for (int k = 0; k < 5; k++)
        {
            threedim[i, j, k] = i * j + k;
            Console.WriteLine("i = {0}, j={1} and k={2} so i*j+k = {3}", i, j, k, threedim[i, j, k]);
        }
    }
}

struct Person
{
    private int i, a;
    private string f, l, g;

    public void getValues(int ID, string fname, string lname, int age, string gender)
    {
        i = ID;
        f = fname;
        l = lname;
        a = age;
        g = gender;
    }

    public void displayValues()
    {
        Console.WriteLine("{0} {1} is ID {2}, age = {3} and gender = {4}", f, l, i, a, g);
    }
}

myArray[0].displayValues(); // please note how the array element has acess to struct method

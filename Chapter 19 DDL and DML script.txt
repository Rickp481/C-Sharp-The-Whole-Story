use model;
go

create table Textbooks
(
	Textbook_ID		int				primary key,
	Title			varchar(45)		not null,
	Publisher		varchar(45)		not null,
	Price			decimal(6,2)	not null
);

create table Departments
(
	Department		varchar(45)		primary key
);

create table Enrollment_Statuses
(
	Enrollment_Status	varchar(45)	primary key
);

create table Instructors
(
	Instructor_ID			int,
	First_Name				varchar(45) not null,
	Last_Name				varchar(45) not null,
	Department				varchar(45) not null,
	Department_Head			int,
	constraint Instructors_pk primary key (Instructor_ID),
	constraint Instructors_fk0 foreign key (Department) references Departments(Department),
	constraint Instructors_fk1 foreign key (Department_Head) references Instructors(Instructor_ID)
);
go

create table Students
(
	Student_ID				int			primary key,
	First_Name				varchar(45) not null,
	Last_Name				varchar(45) not null,
	Phone					varchar(15) not null,
	Email					varchar(45) not null,
	Department_Major		varchar(45) default null,
	Department_Minor		varchar(45) default null,
	constraint Students_fk0 foreign key (Department_Major) references Departments(Department),
	constraint Students_fk1 foreign key (Department_Minor) references Departments(Department)
);
go

create table Courses
(
	Course_ID				int			primary key,
	Title					varchar(45)	not null,
	Department				varchar(45) not null,
	constraint Courses_fk0 (Department) references Departments(Department)
);


create table Instructor_Courses
(
	Instructor_ID			int,
	Course_ID				int,
	Textbook_ID				int,
	constraint Instructor_Courses_pk primary key (Instructor_ID, Course_ID, Textbook_ID),
	constraint Instructor_Courses_fk0 foreign key (Instructor_ID) references Instructors(Instructor_ID),
	constraint Instructor_Courses_fk1 foreign key (Course_ID) references Courses(Course_ID),
	constraint Instructor_Courses_fk2 foreign key (Textbook_ID) references Textbooks(Textbook_ID)
);

create table Student_Courses
(
	Student_ID				int,
	Course_ID				int,
	Enrollment_Status		varchar(45) not null,
	constraint Student_Courses_pk primary key (Student_ID, Course_ID),
	constraint Student_Courses_fk0 foreign key (Student_ID) references Students(Student_ID),
	constraint Student_Courses_fk1 foreign key (Course_ID) references Courses(Course_ID),
	constraint Student_Courses_fk2 foreign key (Enrollment_Status) references Enrollment_Statuses(Enrollment_Status)
);
go

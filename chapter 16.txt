        private void MC_test_driver(Item[] my_items)
        {
            // randomize the questions
            int[] ints = new int[my_items.Length];
            int ints_index = -1, current_random = -1;
            Random rnd = new Random();

            for (int i = 0; i < my_items.Length; i++)
            {
                current_random = rnd.Next(my_items.Length);
                bool found_it = false;
                for (int j = 0; j < ints.Length; j++) 
                    if (ints[j] == current_random) found_it = true;
                if (found_it == false)
                {
                    ints_index += 1;
                    ints[ints_index] = current_random;
                }
            }
            Globals.my_item_global = my_items[0];
            Globals.my_ints_global = ints;
            Globals.my_current_index = 0;

            Load_MC_Question_Data();
        }

        private void setup_MC_question_data()
        {
            // start by unchecking all RadioButtons
            foreach (Control c in this.Controls)
            {
                if (c is GroupBox)
                {
                    foreach(Control c2 in c.Controls)
                    {
                        // all these are RadioButtons
                        RadioButton rb = (RadioButton)c2;
                        rb.Checked = false;
                    }
                }
                else if (c is TextBox)
                {
                    TextBox textBox = (TextBox)c;
                    textBox.Visible = true;
                }
            }

            if (Globals.my_current_index == 0)
            {
                foreach (Control c in this.Controls)
                {
                    if (c is Button)
                    {
                        Button button = (Button)c;
                        if (button.Text == "Previous") button.Visible = false;
                    }
                }

            }
            else if (Globals.my_current_index ==Globals.my_ints_global.Length)
            {
                foreach(Control c  in this.Controls)
                {
                    if (c is Button)
                    {
                        Button button = (Button)c;
                        if (button.Text == "Next") button.Visible = false;
                    }
                }
            }
        }

        private void timer1_Tick (object sender, EventArgs e)
        {
            int time_in_minutes, time_in_seconds;
            string time_in_minutes_s, time_in_seconds_s;

            time_in_minutes = int.Parse((Digital_Clock.Text).Substring(0, 2));
            time_in_seconds = int.Parse((Digital_Clock.Text).Substring(3, 2));

            // if time has expired close the form
            if (up_or_down == "Up" && ((time_in_minutes * 60) + time_in_seconds + 1) >= duration_in_secs) this.Close();
            else if (up_or_down == "Down" && time_in_minutes == 0 && time_in_seconds == 0) this.Close();

            // change Digital_Clock
            else if (up_or_down == "Up" && time_in_seconds == 59) // end of a minute
            {
                time_in_seconds = 0;
                time_in_minutes += 1;
            }
            else if (time_in_seconds == 0) // we are going down so decrement the minute
            {
                time_in_seconds = 0;
                time_in_minutes -= 1;
            }
            else // just decrement seconds
                time_in_seconds -= 1;

            // now we update the display
            time_in_minutes_s = time_in_minutes.ToString();
            if (time_in_minutes_s.Length == 1) time_in_minutes_s = "0" + time_in_minutes_s;
            time_in_seconds_s = time_in_seconds.ToString();
            if (time_in_seconds_s.Length == 1) time_in_seconds_s = "0" + time_in_seconds_s;
            Digital_Clock.Text = time_in_minutes_s + ":" + time_in_seconds_s;
        }

        private void setup_timer()
        {
            if (duration_in_secs == 0) // this course is not timed
            {
                timer1.Stop();
                Digital_Clock.Visible = false;
            }
            else if (up_or_down == "Up")
            {
                Digital_Clock.Text = "00:00";
                Digital_Clock.Visible=true;
                timer1.Start();
            }
            else // up_or_down = "Down"
            {
                string time_in_minutes, time_in_seconds;

                time_in_minutes = (duration_in_secs / 60).ToString();
                time_in_seconds = (duration_in_secs % 60).ToString();

                if (time_in_seconds == "0") time_in_seconds="00"; // just to make it look nicer

                Digital_Clock.Text = time_in_minutes + ":" + time_in_seconds;
                Digital_Clock.Visible=true;
                timer1.Start();
            }
        }
